CREATE TEMP TRIGGER IF NOT EXISTS pushUploaded
AFTER UPDATE OF uploadedAt ON itemsToUpload
WHEN NEW.uploadedAt > -1
BEGIN
  UPDATE moz_bookmarks SET syncStatus = 2 WHERE guid = NEW.guid;
  DELETE FROM moz_bookmarks_deleted WHERE guid = NEW.guid;
END;

DROP TRIGGER IF EXISTS pushUploaded;
