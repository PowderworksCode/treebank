pub fn f(buf: []u8) usize {
    const n = blk: {
        break :blk buf.len;
    };
    outer: while (true) {
        for (buf) |c| {
            if (c == 0) continue :outer;
        }
        break;
    }
    return n;
}

pub fn g(a: Allocator) !void {
    errdefer |e| report(e);
    const buf = try a.alloc(u8, 16);
    const n = parse(buf) catch |err| switch (err) {
        error.Invalid => return err,
        else => unreachable,
    };
    const m = maybe() orelse 0;
    _ = n + m;
}
